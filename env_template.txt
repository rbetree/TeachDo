# ====================================================================
# 根目录 .env 统一模型配置（精简版）
#
# 目标：只按“用途 + 协议类型”配置一套通用字段：
#   *_TYPE       - 协议类型（如：openai、google、claude、ollama 等）
#   *_BASE_URL   - 可选，自托管或 OpenAI 兼容服务的 base_url
#   *_API_KEY    - 对应服务的 API Key
#   *_MODEL      - 模型名称
#
# 具体支持的 TYPE 取值及模型示例，参考 doc/ai/CUSTOM_MODEL.md。
# ====================================================================

# 大纲生成主模型
# 协议类型 TYPE: openai | google | claude | ollama | vllm | local_openai
# 说明：
#   - 使用 OpenAI 官方：   TYPE=openai, BASE_URL 留空，API_KEY=OpenAI Key
#   - 使用 DeepSeek：      TYPE=openai, BASE_URL=https://api.deepseek.com/v1
#   - 使用 阿里 DashScope：TYPE=openai, BASE_URL=https://dashscope.aliyuncs.com/compatible-mode/v1
#   - 使用 豆包：          TYPE=openai, BASE_URL=https://ark.cn-beijing.volces.com/api/v3
OUTLINE_TYPE=openai
OUTLINE_BASE_URL=https://dashscope.aliyuncs.com/compatible-mode/v1
OUTLINE_API_KEY=sk-your_outline_api_key
OUTLINE_MODEL=qwen-turbo-latest

# 幻灯片内容生成模型（同样按“协议类型 + BASE_URL”配置）
PPT_WRITER_TYPE=openai
PPT_WRITER_BASE_URL=https://dashscope.aliyuncs.com/compatible-mode/v1
PPT_WRITER_API_KEY=sk-your_writer_api_key
PPT_WRITER_MODEL=qwen-turbo-latest

# 幻灯片内容校对模型
PPT_CHECKER_TYPE=openai
PPT_CHECKER_BASE_URL=https://dashscope.aliyuncs.com/compatible-mode/v1
PPT_CHECKER_API_KEY=sk-your_checker_api_key
PPT_CHECKER_MODEL=qwen-turbo-latest

# 知识库向量嵌入模型

EMBEDDING_API_KEY=your_embedding_api_key
